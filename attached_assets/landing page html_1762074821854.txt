                }


                // تغيير الزرار الذكي
                if (pricingSection && smartCta) {
                    const pricingTop = pricingSection.offsetTop - 100; 
                    
                    if (window.scrollY > pricingTop) {
                        // العميل عدّى الأسعار
                        if (smartCta.href.includes("#pricing")) { 
                            smartCta.textContent = "💳 اشتري الباقة الأكثر شيوعاً";
                            smartCta.href = "#buy-medium"; 
                            smartCta.classList.remove("cta-primary"); 
                            smartCta.classList.add("cta-blue"); 
                            smartCta.classList.add("open-in-popup"); 
                            smartCta.dataset.url = "https://example.com/buy-medium";
                            smartCta.dataset.title = "شراء باكدج النمو";
                        }
                    } else {
                        // العميل لسه فوق الأسعار
                        if (!smartCta.href.includes("#pricing")) { 
                            smartCta.textContent = "ابدأ دلوقتي";
                            smartCta.href = "#pricing"; 
                            smartCta.classList.add("cta-primary"); 
                            smartCta.classList.remove("cta-blue"); 
                            smartCta.classList.remove("open-in-popup"); 
                            delete smartCta.dataset.url;
                            delete smartCta.dataset.title;
                        }
                    }
                }
            });


            // --- كود تأثير 3D Tilt ---
            const tiltElements = document.querySelectorAll('.tilt-card');
            VanillaTilt.init(tiltElements, {
                max: 5,
                speed: 400,
                perspective: 1000,
                glare: true,
                "max-glare": 0.1
            });


            // --- كود الأسئلة الشائعة (FAQ) ---
            const allDetails = document.querySelectorAll('.faq-item');
            allDetails.forEach(details => {
                details.addEventListener('toggle', (event) => {
                    if (details.open) {
                        allDetails.forEach(d => {
                            if (d !== details && d.open) {
                                d.removeAttribute('open');
                            }
                        });
                    }
                });
            });


            // --- (إضافة) كود الأيقونات المغناطيسية ---
            const magneticIcons = document.querySelectorAll('.magnetic-icon');
            magneticIcons.forEach(icon => {
                icon.addEventListener('mousemove', (e) => {
                    const rect = icon.getBoundingClientRect();
                    const x = (e.clientX - rect.left - rect.width / 2) * 0.3;
                    const y = (e.clientY - rect.top - rect.height / 2) * 0.3;
                    icon.style.transform = `translate(${x}px, ${y}px)`;
                });
                icon.addEventListener('mouseleave', () => {
                    icon.style.transform = 'translate(0, 0)';
                });
            });


        });
    </script>
